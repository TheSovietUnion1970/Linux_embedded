${CC_x86_for_arm}gcc --version

# inside ~/BBB/linux-stable-rcn-ee
cd linux-stable-rcn-ee
make ARCH=arm CROSS_COMPILE=${CC_x86_for_arm} -j$(nproc) clean

make ARCH=arm CROSS_COMPILE=${CC_x86_for_arm} olddefconfig
# (CHECK) - CONFIG_GPIO_OMAP, CONFIG_MODULES=, CONFIG_LOCALVERSION

make ARCH=arm CROSS_COMPILE=${CC_x86_for_arm} -j$(nproc) zImage
# (CHECK) - ls -lh ./arch/arm/boot/zImage

make ARCH=arm CROSS_COMPILE=${CC_x86_for_arm} -j$(nproc) modules
# (CHECK) - ls -lh ./drivers/gpio/gpio-omap.ko

sudo rm -rf /tmp/modules
make ARCH=arm CROSS_COMPILE=${CC_x86_for_arm} INSTALL_MOD_PATH=/tmp/modules  modules_install
# (CHECK) - ls -lh /tmp/modules/lib/modules/5.15.177+/kernel/drivers/gpio/gpio-omap.ko.xz

# copy everything from /tmp/modules/lib/modules/5.15.177+/ -> /media/rootfs/lib/modules/5.15.177+/
sudo rm -rf /media/rootfs/lib/modules/5.15.177+
sudo mkdir -p /media/rootfs/lib/modules/5.15.177+
sudo cp -a /tmp/modules/lib/modules/5.15.177+/* /media/rootfs/lib/modules/5.15.177+/
# (CHECK) - ls -lh /media/rootfs/lib/modules/5.15.177+/kernel/drivers/gpio/gpio-omap.ko.xz

# Uncompress the .ko.xz files to .ko and delete Compressed ko
cd /media/rootfs/lib/modules/5.15.177+
find . -type f -name "*.ko.xz" -exec sudo unxz {} \;
# (CHECK) - ls -lh /media/rootfs/lib/modules/5.15.177+/kernel/drivers/gpio/gpio-omap.ko
# (CHECK) - find /media/rootfs/lib/modules/5.15.177+ -type f -name "*.ko.xz"

		=== initramfs ===
# Creates modules.dep and other dependency files needed for modprobe to load modules correctly
// sudo chroot /media/rootfs/ /bin/bash -c "depmod 5.15.177+"
sudo depmod -b /media/rootfs/ 5.15.177+
# (CHECK) - ls -l /media/rootfs/lib/modules/5.15.177+/ => modules.alias, modules.dep, modules.symbols, ...

mkdir -p /tmp/initramfs-tools
cd /tmp/initramfs-tools
apt download initramfs-tools:armhf

sudo mkdir -p /media/rootfs/etc/initramfs-tools
echo "MODULES=most" | sudo tee /media/rootfs/etc/initramfs-tools/initramfs.conf
sudo mkinitramfs -o /media/rootfs/boot/initrd.img-5.15.177+ \
                 -r /media/rootfs \
                 -d /media/rootfs/etc/initramfs-tools \
                 5.15.177+
