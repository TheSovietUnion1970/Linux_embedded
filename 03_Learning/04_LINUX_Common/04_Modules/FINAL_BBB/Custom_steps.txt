wget -c https://mirrors.edge.kernel.org/pub/tools/crosstool/files/bin/x86_64/11.3.0/x86_64-gcc-11.3.0-nolibc-arm-linux-gnueabi.tar.xz
tar -xf x86_64-gcc-11.3.0-nolibc-arm-linux-gnueabi.tar.xz


export CC=`pwd`/gcc-11.3.0-nolibc/arm-linux-gnueabi/bin/arm-linux-gnueabi-
sudo apt-get install flex bison build-essential -> all necessary tools (gcc, flex, ...)
		=== === ===

make ARCH=arm CROSS_COMPILE=${CC} omap2plus_defconfig -> create .config
nano .config

# CONFIG_MODULES is not set -> CONFIG_MODULES=y

make ARCH=arm CROSS_COMPILE=${CC} olddefconfig -> Sync .config with new options post-upgrade

make ARCH=arm CROSS_COMPILE=${CC} -j$(nproc) zImage
# make ARCH=arm CROSS_COMPILE=${CC} dtbs -> use default dtb
make ARCH=arm CROSS_COMPILE=${CC} -j$(nproc) modules

		=== === ===

# installs the compiled modules into /tmp/modules/lib/modules/5.15.177+/ (*ko.xz)
sudo rm -rf /tmp/modules
# (create /tmp/modules/lib/modules/5.15.177+)
make ARCH=arm CROSS_COMPILE=arm-linux-gnueabihf- INSTALL_MOD_PATH=/tmp/modules modules_install

# copy everything from /tmp/modules/lib/modules/5.15.177+/ -> /media/rootfs/lib/modules/5.15.177+/
# Copies all kernel modules (including gpio-omap.ko.xz, build, kernel, source, modules.order) to the SD card’s root filesystem.
sudo rm -rf /media/rootfs/lib/modules/5.15.177+
sudo mkdir -p /media/rootfs/lib/modules/5.15.177+
sudo cp -a /tmp/modules/lib/modules/5.15.177+/* /media/rootfs/lib/modules/5.15.177+/

# Uncompress the .ko.xz files to .ko
cd /media/rootfs/lib/modules/5.15.177+
find . -type f -name "*.ko.xz" -exec sudo unxz {} \;

# (CHECK) - 
ls /media/rootfs/lib/modules/5.15.177+/kernel/drivers/gpio/gpio-omap.ko
find /media/rootfs/lib/modules/5.15.177+ -type f -name "*.ko.xz" -> nothing happens

		=== initramfs ===

# Creates modules.dep and other dependency files needed for modprobe to load modules correctly
sudo chroot /media/rootfs/ /bin/bash -c "depmod 5.15.177+"

# (OPTIONAL) - Force Include gpio_omap.ko in the Initramfs if not
echo "gpio_omap" | sudo tee -a /media/rootfs/etc/initramfs-tools/modules

# (OPTIONAL) - Install initramfs-tools (if not already installed):
sudo chroot /media/rootfs/ /bin/bash -c "apt update && apt install -y initramfs-tools"

# Generate the initramfs (output: initrd.img-5.15.177-bone43)
# sudo chroot /media/rootfs/ /bin/bash -c "depmod 5.15.177-bone43"
# may get warning: W: Couldn't identify type of root file system for fsck hook
sudo chroot /media/rootfs/ /bin/bash -c "mkinitramfs -o /boot/initrd.img-5.15.177-bone43 5.15.177+"

# (CHECK) - Check that gpio-omap.ko is in the initramfs:
zcat /media/rootfs/boot/initrd.img-5.15.177-bone43 | cpio -t | grep gpio-omap
