static u32 musb_default_fifo_offset(u8 epnum)
{
	return 0x20 + (epnum * 4);
} (musb_core.c) => entry takes a word (4 bytes) at a time

static u32 tusb_fifo_offset(u8 epnum)
{
	return 0x200 + (epnum * 0x20);
}

static u32 tusb_ep_offset(u8 epnum, u16 offset)
{
	return 0x10 + offset;
}
