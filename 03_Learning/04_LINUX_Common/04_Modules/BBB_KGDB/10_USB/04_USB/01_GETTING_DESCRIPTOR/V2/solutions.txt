void print_arr(u8* d, u16 len, u8* name) {
    char buf[256];
    int pos = 0;
    u8 i;

    pos += snprintf(buf + pos, sizeof(buf) - pos, "%s[%d] = \n", name, len);
    for (i = 0; i < 6 && pos < sizeof(buf) - 20; i++) {
        pos += snprintf(buf + pos, sizeof(buf) - pos, "i = %d\n", i);
    }
    printk("%s", buf);
}
