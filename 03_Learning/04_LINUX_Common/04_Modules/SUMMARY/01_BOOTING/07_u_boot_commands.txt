kernel_addr_r=0x82000000                                                        
fdt_addr_r=0x88000000                                                           
ramdisk_addr_r=0x88080000  

3. bootdelay, bootcmd

4. mmc
mmc dev 0	-> switch to dev 0 (SD card)
mmc part	-> show info
ext4ls mmc 0:1	-> show dir
ext4load mmc 0:1 ${kernel_addr_r} /boot/vmlinuz-5.15.177
